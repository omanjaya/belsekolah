# Bel Sekolah — Rencana 1: Inti Bel (Backend Core) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Binary Go `belsekolah` yang membaca jadwal dari SQLite dan membunyikan bel tepat waktu lewat `mpv`, lengkap dengan profil, periode khusus, libur, nada bawaan, nada cadangan, dan riwayat.

**Architecture:** Paket `schedule` (logika murni) menentukan bel pada suatu tanggal; `store` menyimpan data di SQLite; `scheduler` berdetak tiap detik dan mengirim permintaan ke `player`, yang memutar audio berurutan via `mpv` dengan antrean dan prioritas darurat. `nada` menyintesis suara bawaan (WAV) tanpa file eksternal; `bawaan` mengisi data awal. Rencana ini belum memiliki web/API — itu Rencana 2.

**Tech Stack:** Go 1.27, `modernc.org/sqlite` (pure Go, tanpa cgo — memudahkan cross-compile ke linux/amd64), `log/slog`, `mpv` (eksternal).

**Spec:** `docs/superpowers/specs/2026-10-15-bel-sekolah-design.md`

## Global Constraints

- Module path: `belsekolah`.
- Semua teks yang terlihat pengguna (pesan error, log, nama data awal) berbahasa Indonesia.
- Format tanggal `YYYY-MM-DD` (`schedule.FormatTanggal`), format jam `HH:MM`.
- Hari: 1=Senin … 7=Minggu.
- Toleransi bel terlambat: 60 detik; bel terlewat lebih dari itu tidak disusulkan, dicatat `terlewat`.
- Status riwayat: `berhasil`, `gagal`, `terlewat`, `cadangan`, `dihentikan`. Sumber riwayat: `jadwal`, `manual`, `darurat`.
- Data aplikasi di folder `-data` (default `/var/lib/belsekolah`): `bel.db`, `audio/`, `cadangan.wav`.
- Tanpa cgo: driver SQLite harus `modernc.org/sqlite`.
- Paket `schedule` tidak boleh mengimpor paket lain dari proyek ini dan tidak boleh membaca jam sistem, file, atau database.

// Package schedule menentukan bel apa saja yang berbunyi pada suatu tanggal.
// Paket ini murni: tidak membaca jam sistem, database, maupun file.
package schedule

import (
	"fmt"
	"sort"
	"time"
)

// FormatTanggal adalah format tanggal yang dipakai di seluruh aplikasi.
const FormatTanggal = "2006-01-02"

// Jadwal adalah satu bel pada hari tertentu dalam sebuah profil.
type Jadwal struct {
	ID         int64
	ProfilID   int64
	Hari       int    // 1=Senin ... 7=Minggu
	Jam        string // "HH:MM"
	JenisBelID int64
	Label      string
}

// Periode membuat rentang tanggal memakai profil tertentu.
type Periode struct {
	ID         int64
	Mulai      string // YYYY-MM-DD, inklusif
	Selesai    string // YYYY-MM-DD, inklusif
	ProfilID   int64
	Keterangan string
}

// Libur adalah rentang tanggal tanpa bel terjadwal.
type Libur struct {
	ID         int64
	Mulai      string // YYYY-MM-DD, inklusif
	Selesai    string // YYYY-MM-DD, inklusif
	Keterangan string
}

// Data adalah seluruh data yang dibutuhkan untuk menentukan bel.
type Data struct {
	ProfilDefaultID int64
	Periode         []Periode
	Libur           []Libur
	Jadwal          []Jadwal
}

// Bel adalah satu bel yang harus berbunyi pada waktu tertentu.
type Bel struct {
	JadwalID   int64
	Waktu      time.Time
	JenisBelID int64
	Label      string
}

// Hasil adalah bel pada suatu tanggal beserta profil yang dipakai.
type Hasil struct {
	ProfilID int64  // 0 jika libur
	Libur    *Libur // nil jika bukan hari libur
	Bel      []Bel  // terurut berdasarkan waktu
}

// ParseJam mengurai jam berformat HH:MM.
func ParseJam(s string) (jam, menit int, err error) {
	t, err := time.Parse("15:04", s)
	if err != nil || len(s) != 5 {
		return 0, 0, fmt.Errorf("jam tidak valid %q, gunakan format HH:MM", s)
	}
	return t.Hour(), t.Minute(), nil
}

// Hari mengembalikan 1=Senin ... 7=Minggu.
func Hari(t time.Time) int {
	h := int(t.Weekday())
	if h == 0 {
		return 7
	}
	return h
}

func dalamRentang(tgl, mulai, selesai string) bool {
	return mulai <= tgl && tgl <= selesai
}

// BelPadaTanggal menerapkan aturan: libur → periode khusus → profil default.
// Waktu bel memakai zona waktu dari tgl.
func BelPadaTanggal(tgl time.Time, d Data) Hasil {
	kunci := tgl.Format(FormatTanggal)
	for i := range d.Libur {
		if dalamRentang(kunci, d.Libur[i].Mulai, d.Libur[i].Selesai) {
			l := d.Libur[i]
			return Hasil{Libur: &l}
		}
	}

	profil := d.ProfilDefaultID
	for _, p := range d.Periode {
		if dalamRentang(kunci, p.Mulai, p.Selesai) {
			profil = p.ProfilID
			break
		}
	}

	hari := Hari(tgl)
	var bel []Bel
	for _, j := range d.Jadwal {
		if j.ProfilID != profil || j.Hari != hari {
			continue
		}
		h, m, err := ParseJam(j.Jam)
		if err != nil {
			continue
		}
		bel = append(bel, Bel{
			JadwalID:   j.ID,
			Waktu:      time.Date(tgl.Year(), tgl.Month(), tgl.Day(), h, m, 0, 0, tgl.Location()),
			JenisBelID: j.JenisBelID,
			Label:      j.Label,
		})
	}
	sort.Slice(bel, func(a, b int) bool {
		if !bel[a].Waktu.Equal(bel[b].Waktu) {
			return bel[a].Waktu.Before(bel[b].Waktu)
		}
		return bel[a].JadwalID < bel[b].JadwalID
	})
	return Hasil{ProfilID: profil, Bel: bel}
}

//go:embed schema.sql
var skema string

var (
	ErrTidakAda             = errors.New("data tidak ditemukan")
	ErrPeriodeTumpangTindih = errors.New("periode khusus bertabrakan dengan periode lain")
)

// KunciProfilDefault adalah kunci pengaturan berisi ID profil default.
const KunciProfilDefault = "profil_default_id"

// Sumber bel pada riwayat.
const (
	SumberJadwal  = "jadwal"
	SumberManual  = "manual"
	SumberDarurat = "darurat"
)

type Store struct {
	db *sql.DB
}

// Open membuka (atau membuat) database dan menjalankan migrasi skema.
func Open(path string) (*Store, error) {
	dsn := "file:" + path + "?_pragma=foreign_keys(1)&_pragma=journal_mode(WAL)&_pragma=busy_timeout(5000)"
	db, err := sql.Open("sqlite", dsn)
	if err != nil {
		return nil, err
	}
	// Satu koneksi: SQLite hanya punya satu penulis, dan ini menghindari SQLITE_BUSY.
	db.SetMaxOpenConns(1)
	if _, err := db.Exec(skema); err != nil {
		db.Close()
		return nil, fmt.Errorf("migrasi skema: %w", err)
	}
	return &Store{db: db}, nil
}

func (s *Store) Close() error {
	return s.db.Close()
}

func (s *Store) insert(ctx context.Context, query string, args ...any) (int64, error) {
	res, err := s.db.ExecContext(ctx, query, args...)
	if err != nil {
		return 0, err
	}
	return res.LastInsertId()
}

// GetSetting mengembalikan ErrTidakAda jika kunci belum pernah disimpan.
func (s *Store) GetSetting(ctx context.Context, kunci string) (string, error) {
	var nilai string
	err := s.db.QueryRowContext(ctx, `SELECT nilai FROM pengaturan WHERE kunci = ?`, kunci).Scan(&nilai)
	if errors.Is(err, sql.ErrNoRows) {
		return "", ErrTidakAda
	}
	return nilai, err
}

func (s *Store) SetSetting(ctx context.Context, kunci, nilai string) error {
	_, err := s.db.ExecContext(ctx,
		`INSERT INTO pengaturan (kunci, nilai) VALUES (?, ?)
		 ON CONFLICT (kunci) DO UPDATE SET nilai = excluded.nilai`, kunci, nilai)
	return err
}

import (
	"context"
	"database/sql"
	"errors"
	"fmt"
	"time"

// Audio adalah satu file suara. File relatif terhadap folder audio.
type Audio struct {
	ID       int64
	Nama     string
	Sumber   string // bawaan, upload, tts
	File     string
	DurasiMS int64
	TeksTTS  string
	SuaraTTS string
	Dibuat   time.Time
}

type JenisBel struct {
	ID    int64
	Nama  string
	Warna string
}

const kolomAudio = `a.id, a.nama, a.sumber, a.file, a.durasi_ms, a.teks_tts, a.suara_tts, a.dibuat`

type pemindai interface {
	Scan(dest ...any) error
}

func scanAudio(r pemindai) (Audio, error) {
	var a Audio
	var dibuat string
	err := r.Scan(&a.ID, &a.Nama, &a.Sumber, &a.File, &a.DurasiMS, &a.TeksTTS, &a.SuaraTTS, &dibuat)
	if errors.Is(err, sql.ErrNoRows) {
		return Audio{}, ErrTidakAda
	}
	if err != nil {
		return Audio{}, err
	}
	if a.Dibuat, err = time.Parse(time.RFC3339, dibuat); err != nil {
		return Audio{}, fmt.Errorf("waktu dibuat audio %d tidak valid: %w", a.ID, err)
	}
	return a, nil
}

func (s *Store) CreateAudio(ctx context.Context, a Audio) (int64, error) {
	return s.insert(ctx,
		`INSERT INTO audio (nama, sumber, file, durasi_ms, teks_tts, suara_tts, dibuat)
		 VALUES (?, ?, ?, ?, ?, ?, ?)`,
		a.Nama, a.Sumber, a.File, a.DurasiMS, a.TeksTTS, a.SuaraTTS, a.Dibuat.Format(time.RFC3339))
}

func (s *Store) GetAudio(ctx context.Context, id int64) (Audio, error) {
	return scanAudio(s.db.QueryRowContext(ctx, `SELECT `+kolomAudio+` FROM audio a WHERE a.id = ?`, id))
}

func (s *Store) FindAudio(ctx context.Context, sumber, nama string) (Audio, error) {
	return scanAudio(s.db.QueryRowContext(ctx,
		`SELECT `+kolomAudio+` FROM audio a WHERE a.sumber = ? AND a.nama = ? ORDER BY a.id LIMIT 1`, sumber, nama))
}

func (s *Store) CreateJenisBel(ctx context.Context, nama, warna string) (int64, error) {
	return s.insert(ctx, `INSERT INTO jenis_bel (nama, warna) VALUES (?, ?)`, nama, warna)
}

func (s *Store) ListJenisBel(ctx context.Context) ([]JenisBel, error) {

	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []JenisBel
	for rows.Next() {
		var j JenisBel
		if err := rows.Scan(&j.ID, &j.Nama, &j.Warna); err != nil {
			return nil, err
		}
		out = append(out, j)
	}
	return out, rows.Err()
}

// SetUrutanBel mengganti seluruh urutan audio milik sebuah jenis bel.
func (s *Store) SetUrutanBel(ctx context.Context, jenisBelID int64, audioIDs []int64) error {
	tx, err := s.db.BeginTx(ctx, nil)
	if err != nil {
		return err
	}
	defer tx.Rollback()
	if _, err := tx.ExecContext(ctx, `DELETE FROM urutan_bel WHERE jenis_bel_id = ?`, jenisBelID); err != nil {
		return err
	}
	for i, audioID := range audioIDs {
		if _, err := tx.ExecContext(ctx,
			`INSERT INTO urutan_bel (jenis_bel_id, urutan, audio_id) VALUES (?, ?, ?)`,
			jenisBelID, i+1, audioID); err != nil {
			return err
		}
	}
	return tx.Commit()
}

// UrutanBel mengembalikan audio milik jenis bel sesuai urutan putar.
func (s *Store) UrutanBel(ctx context.Context, jenisBelID int64) ([]Audio, error) {

		`SELECT `+kolomAudio+` FROM urutan_bel u JOIN audio a ON a.id = u.audio_id
		 WHERE u.jenis_bel_id = ? ORDER BY u.urutan`, jenisBelID)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []Audio
	for rows.Next() {
		a, err := scanAudio(rows)
		if err != nil {
			return nil, err
		}
		out = append(out, a)
	}
	return out, rows.Err()
}

	}

	if err != nil {
		return fmt.Errorf("tanggal selesai tidak valid %q, gunakan format YYYY-MM-DD", selesai)

// Package nada menyintesis suara bel bawaan (WAV) tanpa file eksternal,
// sehingga aplikasi selalu punya suara walau folder audio kosong atau rusak.
package nada

import (
	"encoding/binary"
	"math"
)

const sampleRate = 44100

// Bawaan adalah satu suara bawaan yang dipasang ke folder audio.
type Bawaan struct {
	Nama string
	File string
	Data func() []byte
}

var Daftar = []Bawaan{
	{Nama: "Ding Dong", File: "bawaan-ding-dong.wav", Data: DingDong},
	{Nama: "Ding Dong Panjang", File: "bawaan-ding-dong-panjang.wav", Data: DingDongPanjang},
	{Nama: "Sirene", File: "bawaan-sirene.wav", Data: Sirene},
}

type not struct {
	frek   float64
	durasi float64 // detik
}

// DingDong: dua nada turun E5 → C5.
func DingDong() []byte {
	return wav(render([]not{{659.25, 0.7}, {523.25, 1.1}}, 3))
}

// DingDongPanjang: empat nada naik C5 → E5 → G5 → C6.
func DingDongPanjang() []byte {
	return wav(render([]not{{523.25, 0.6}, {659.25, 0.6}, {783.99, 0.6}, {1046.5, 1.4}}, 2.5))
}

// Sirene: frekuensi naik-turun 600–1200 Hz, tiga siklus.
func Sirene() []byte {
	const durasi = 4.5
	const siklus = 1.5
	s := make([]int16, int(durasi*sampleRate))
	fase := 0.0
	for i := range s {
		t := float64(i) / sampleRate
		posisi := math.Mod(t, siklus) / siklus
		frek := 600 + 600*(1-math.Abs(2*posisi-1))
		fase += 2 * math.Pi * frek / sampleRate
		s[i] = int16(math.Sin(fase) * 0.7 * math.MaxInt16)
	}
	return wav(s)
}

// Cadangan diputar jika file audio sebuah bel hilang atau rusak.
func Cadangan() []byte {
	return DingDong()
}

// DurasiMS menghitung durasi WAV hasil paket ini.
func DurasiMS(b []byte) int64 {
	if len(b) <= 44 {
		return 0
	}
	sampel := int64(len(b)-44) / 2
	return sampel * 1000 / sampleRate
}

// render membuat nada lonceng: nada dasar + harmonik kedua, peluruhan eksponensial.
func render(nots []not, peluruhan float64) []int16 {
	var out []int16
	for _, n := range nots {
		jumlah := int(n.durasi * sampleRate)
		for i := 0; i < jumlah; i++ {
			t := float64(i) / sampleRate
			amp := math.Exp(-peluruhan * t)
			if t < 0.005 {
				amp *= t / 0.005 // serangan halus agar tidak "klik"
			}
			v := (math.Sin(2*math.Pi*n.frek*t) + 0.3*math.Sin(4*math.Pi*n.frek*t)) / 1.3
			out = append(out, int16(v*amp*0.8*math.MaxInt16))
		}
	}
	return out
}

func wav(s []int16) []byte {
	dataLen := len(s) * 2
	b := make([]byte, 44+dataLen)
	copy(b[0:], "RIFF")
	binary.LittleEndian.PutUint32(b[4:], uint32(36+dataLen))
	copy(b[8:], "WAVE")
	copy(b[12:], "fmt ")
	binary.LittleEndian.PutUint32(b[16:], 16)           // ukuran chunk fmt
	binary.LittleEndian.PutUint16(b[20:], 1)            // PCM
	binary.LittleEndian.PutUint16(b[22:], 1)            // mono
	binary.LittleEndian.PutUint32(b[24:], sampleRate)   // sample rate
	binary.LittleEndian.PutUint32(b[28:], sampleRate*2) // byte rate
	binary.LittleEndian.PutUint16(b[32:], 2)            // block align
	binary.LittleEndian.PutUint16(b[34:], 16)           // bit per sampel
	copy(b[36:], "data")
	binary.LittleEndian.PutUint32(b[40:], uint32(dataLen))
	for i, v := range s {
		binary.LittleEndian.PutUint16(b[44+i*2:], uint16(v))
	}
	return b
}

// Bel darurat memotong suara yang sedang diputar dan mengosongkan antrean.
package player

import (
	"context"
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"sync"
)

type Status string

// Nilai Status sama dengan status pada riwayat bel.
const (
	Berhasil   Status = "berhasil"
	Cadangan   Status = "cadangan"
	Gagal      Status = "gagal"
	Dihentikan Status = "dihentikan"
)

type Prioritas int

const (
	Biasa Prioritas = iota
	Darurat
)

// Permintaan adalah satu bel: beberapa file diputar berurutan.
// Selesai (boleh nil) dipanggil tepat sekali dari goroutine player.
type Permintaan struct {
	Files     []string
	Prioritas Prioritas
	Selesai   func(status Status, pesan string)
}

// Runner memutar satu file sampai selesai, atau berhenti saat ctx dibatalkan.
type Runner interface {
	Run(ctx context.Context, file string, volume int) error
}

type Player struct {
	runner   Runner
	cadangan string
	sinyal   chan struct{}

	mu      sync.Mutex
	antrean []Permintaan
	batal   context.CancelFunc // membatalkan permintaan yang sedang diputar
	volume  int
}

func New(runner Runner, fileCadangan string) *Player {
	return &Player{
		runner:   runner,
		cadangan: fileCadangan,
		sinyal:   make(chan struct{}, 1),
		volume:   100,
	}
}

// SetVolume mengatur volume 0–100 untuk permintaan berikutnya.
func (p *Player) SetVolume(v int) {
	v = max(0, min(100, v))
	p.mu.Lock()
	p.volume = v
	p.mu.Unlock()
}

func (p *Player) Play(req Permintaan) {
	p.mu.Lock()
	var dibuang []Permintaan
	if req.Prioritas == Darurat {
		dibuang = p.antrean
		p.antrean = []Permintaan{req}
		if p.batal != nil {
			p.batal()
		}
	} else {
		p.antrean = append(p.antrean, req)
	}
	p.mu.Unlock()
	for _, d := range dibuang {
		selesai(d, Dihentikan, "dihentikan oleh bel darurat")
	}
	p.bangunkan()
}

// Stop menghentikan suara yang sedang diputar dan mengosongkan antrean.
func (p *Player) Stop() {
	p.mu.Lock()
	dibuang := p.antrean
	p.antrean = nil
	if p.batal != nil {
		p.batal()
	}
	p.mu.Unlock()
	for _, d := range dibuang {
		selesai(d, Dihentikan, "dihentikan oleh admin")
	}
}

// Jalankan memproses antrean sampai ctx selesai.
func (p *Player) Jalankan(ctx context.Context) {
	for {
		p.mu.Lock()
		if len(p.antrean) == 0 {
			p.mu.Unlock()
			select {
			case <-ctx.Done():
				return
			case <-p.sinyal:
				continue
			}
		}
		req := p.antrean[0]
		p.antrean = p.antrean[1:]
		reqCtx, batal := context.WithCancel(ctx)
		p.batal = batal
		volume := p.volume
		p.mu.Unlock()

		status, pesan := p.putar(reqCtx, req.Files, volume)

		p.mu.Lock()
		p.batal = nil
		p.mu.Unlock()
		batal()
		selesai(req, status, pesan)
	}
}

func (p *Player) bangunkan() {
	select {
	case p.sinyal <- struct{}{}:
	default:
	}
}

func selesai(req Permintaan, status Status, pesan string) {
	if req.Selesai != nil {
		req.Selesai(status, pesan)
	}
}

var peringkat = map[Status]int{Berhasil: 0, Cadangan: 1, Gagal: 2, Dihentikan: 3}

func lebihBuruk(a, b Status) Status {
	if peringkat[b] > peringkat[a] {
		return b
	}
	return a
}

func (p *Player) putar(ctx context.Context, files []string, volume int) (Status, string) {
	if len(files) == 0 {
		return p.putarCadangan(ctx, volume, "jenis bel belum memiliki suara")
	}
	status := Berhasil
	var pesan []string
	for _, f := range files {
		if _, err := os.Stat(f); err != nil {
			st, ps := p.putarCadangan(ctx, volume, fmt.Sprintf("file %s tidak bisa dibaca", filepath.Base(f)))
			status = lebihBuruk(status, st)
			pesan = append(pesan, ps)
			if st == Dihentikan {
				return Dihentikan, strings.Join(pesan, "; ")
			}
			continue
		}
		err := p.runner.Run(ctx, f, volume)
		if err != nil && ctx.Err() == nil {
			err = p.runner.Run(ctx, f, volume) // coba ulang sekali
		}
		if ctx.Err() != nil {
			return Dihentikan, "pemutaran dihentikan"
		}
		if err != nil {
			status = lebihBuruk(status, Gagal)
			pesan = append(pesan, fmt.Sprintf("gagal memutar %s: %v", filepath.Base(f), err))
		}
	}
	return status, strings.Join(pesan, "; ")
}

func (p *Player) putarCadangan(ctx context.Context, volume int, alasan string) (Status, string) {
	err := p.runner.Run(ctx, p.cadangan, volume)
	if ctx.Err() != nil {
		return Dihentikan, alasan
	}
	if err != nil {
		return Gagal, alasan + "; nada cadangan juga gagal: " + err.Error()
	}
	return Cadangan, alasan + ", diganti nada cadangan"
}

	Perintah string // kosong berarti "mpv"
}

func (m Mpv) Run(ctx context.Context, file string, volume int) error {
	perintah := m.Perintah
	if perintah == "" {
		perintah = "mpv"
	}
	cmd := exec.CommandContext(ctx, perintah,
		"--no-video", "--no-terminal", "--really-quiet",
		fmt.Sprintf("--volume=%d", volume),
		"--", file)
	out, err := cmd.CombinedOutput()
	if ctx.Err() != nil {
		return ctx.Err()
	}
	if err != nil {
		return fmt.Errorf("%w: %s", err, strings.TrimSpace(string(out)))
	}
	return nil
}

// Package scheduler memeriksa jadwal setiap detik dan membunyikan bel tepat waktu.
package scheduler

import (
	"context"
	"fmt"
	"log/slog"
	"path/filepath"
	"sync"
	"time"

	"belsekolah/internal/player"
	"belsekolah/internal/schedule"
	"belsekolah/internal/store"
)

// Toleransi: bel yang terlambat lebih dari ini tidak dibunyikan, hanya dicatat terlewat.
const Toleransi = 60 * time.Second

type Clock interface {
	Now() time.Time
}

type JamSistem struct{}

func (JamSistem) Now() time.Time { return time.Now() }

type Sumber interface {
	LoadDataJadwal(ctx context.Context) (schedule.Data, error)
	JadwalTercatat(ctx context.Context, tanggal string) (map[int64]bool, error)
	InsertRiwayat(ctx context.Context, r store.Riwayat) error
	UrutanBel(ctx context.Context, jenisBelID int64) ([]store.Audio, error)
}

type Pemutar interface {
	Play(req player.Permintaan)
}

type Scheduler struct {
	clock    Clock
	sumber   Sumber
	pemutar  Pemutar
	audioDir string
	log      *slog.Logger

	mu        sync.Mutex
	tanggal   string
	hasil     schedule.Hasil
	sudah     map[int64]bool // jadwal hari ini yang sudah dibunyikan/dicatat
	perluMuat bool
	editan    bool // muat ulang karena admin mengubah data
}

func New(clock Clock, sumber Sumber, pemutar Pemutar, audioDir string, log *slog.Logger) *Scheduler {
	return &Scheduler{
		clock:     clock,
		sumber:    sumber,
		pemutar:   pemutar,
		audioDir:  audioDir,
		log:       log,
		perluMuat: true,
	}
}

// Reload meminta data dimuat ulang pada detik berikutnya (dipanggil setelah admin mengubah data).
func (s *Scheduler) Reload() {
	s.mu.Lock()
	s.perluMuat = true
	s.editan = true
	s.mu.Unlock()
}

// Run memanggil Tick setiap detik sampai ctx selesai.
func (s *Scheduler) Run(ctx context.Context) {
	t := time.NewTicker(time.Second)
	defer t.Stop()
	for {
		if err := s.Tick(ctx); err != nil {
			s.log.Error("pemeriksaan jadwal gagal", "err", err)
		}
		select {
		case <-ctx.Done():
			return
		case <-t.C:
		}
	}
}

// Tick memeriksa sekali apakah ada bel yang jatuh tempo.
func (s *Scheduler) Tick(ctx context.Context) error {
	now := s.clock.Now()
	tanggal := now.Format(schedule.FormatTanggal)

	s.mu.Lock()
	defer s.mu.Unlock()

	if s.perluMuat || tanggal != s.tanggal {
		gantiHari := tanggal != s.tanggal
		if err := s.muat(ctx, now, tanggal); err != nil {
			return err
		}
		if s.editan && !gantiHari {
			// Bel yang baru dibuat setelah waktunya lewat tidak dianggap terlewat.
			for _, b := range s.hasil.Bel {
				if now.Sub(b.Waktu) >= Toleransi {
					s.sudah[b.JadwalID] = true
				}
			}
		}
		s.perluMuat, s.editan = false, false
	}

	for _, b := range s.hasil.Bel {
		if s.sudah[b.JadwalID] || now.Before(b.Waktu) {
			continue
		}
		s.sudah[b.JadwalID] = true
		if telat := now.Sub(b.Waktu); telat >= Toleransi {
			s.catat(ctx, b, "terlewat", fmt.Sprintf("terlewat %s", telat.Round(time.Second)))
			continue
		}
		s.bunyikan(ctx, b)
	}
	return nil
}

func (s *Scheduler) muat(ctx context.Context, now time.Time, tanggal string) error {
	data, err := s.sumber.LoadDataJadwal(ctx)
	if err != nil {
		return fmt.Errorf("muat jadwal: %w", err)
	}
	tercatat, err := s.sumber.JadwalTercatat(ctx, tanggal)
	if err != nil {
		return fmt.Errorf("muat riwayat: %w", err)
	}
	if tanggal == s.tanggal {
		// Bel yang sedang diputar belum punya riwayat; pertahankan catatan di memori.
		for id := range s.sudah {
			tercatat[id] = true
		}
	}
	s.tanggal = tanggal
	s.hasil = schedule.BelPadaTanggal(now, data)
	s.sudah = tercatat
	return nil
}

func (s *Scheduler) bunyikan(ctx context.Context, b schedule.Bel) {
	var files []string
	audio, err := s.sumber.UrutanBel(ctx, b.JenisBelID)
	if err != nil {
		// Tanpa file, player memutar nada cadangan.
		s.log.Error("muat urutan bel gagal", "jenis_bel", b.JenisBelID, "err", err)
	}
	for _, a := range audio {
		files = append(files, filepath.Join(s.audioDir, a.File))
	}
	s.log.Info("bel berbunyi", "label", b.Label, "waktu", b.Waktu.Format("15:04"))
	s.pemutar.Play(player.Permintaan{
		Files:     files,
		Prioritas: player.Biasa,
		Selesai: func(status player.Status, pesan string) {
			s.catat(context.Background(), b, string(status), pesan)
		},
	})
}

func (s *Scheduler) catat(ctx context.Context, b schedule.Bel, status, pesan string) {
	err := s.sumber.InsertRiwayat(ctx, store.Riwayat{
		Waktu:      b.Waktu,
		JadwalID:   b.JadwalID,
		JenisBelID: b.JenisBelID,
		Label:      b.Label,
		Sumber:     store.SumberJadwal,
		Status:     status,
		Pesan:      pesan,
	})
	if err != nil {
		s.log.Error("simpan riwayat gagal", "label", b.Label, "err", err)
	}
}

// Package bawaan memasang suara bawaan dan data awal untuk instalasi baru.
package bawaan

import (
	"context"
	"errors"
	"fmt"
	"os"
	"path/filepath"
	"strconv"
	"time"

	"belsekolah/internal/nada"
	"belsekolah/internal/store"
)

type jenisAwal struct {
	nama  string
	warna string
	audio []string // nama audio bawaan, diputar berurutan
}

var jenisBelAwal = []jenisAwal{
	{"Masuk", "#2563eb", []string{"Ding Dong Panjang"}},
	{"Ganti Pelajaran", "#0d9488", []string{"Ding Dong"}},
	{"Istirahat", "#16a34a", []string{"Ding Dong", "Ding Dong"}},
	{"Pulang", "#9333ea", []string{"Ding Dong Panjang", "Ding Dong Panjang"}},
	{"Darurat", "#dc2626", []string{"Sirene", "Sirene"}},
}

// Pasang menulis file suara bawaan yang belum ada, mendaftarkannya ke database,
// dan—hanya jika belum ada profil—membuat profil "Normal" serta jenis bel awal.
func Pasang(ctx context.Context, st *store.Store, audioDir string, now time.Time) error {
	ids := make(map[string]int64)
	for _, b := range nada.Daftar {
		data := b.Data()
		path := filepath.Join(audioDir, b.File)
		if _, err := os.Stat(path); errors.Is(err, os.ErrNotExist) {
			if err := os.WriteFile(path, data, 0o644); err != nil {
				return fmt.Errorf("tulis %s: %w", b.File, err)
			}
		}
		a, err := st.FindAudio(ctx, "bawaan", b.Nama)
		switch {
		case errors.Is(err, store.ErrTidakAda):
			id, err := st.CreateAudio(ctx, store.Audio{
				Nama: b.Nama, Sumber: "bawaan", File: b.File,
				DurasiMS: nada.DurasiMS(data), Dibuat: now,
			})
			if err != nil {
				return fmt.Errorf("daftarkan %s: %w", b.Nama, err)
			}
			ids[b.Nama] = id
		case err != nil:
			return err
		default:
			ids[b.Nama] = a.ID
		}
	}

	n, err := st.CountProfil(ctx)
	if err != nil {
		return err
	}
	if n > 0 {
		return nil
	}

	for _, j := range jenisBelAwal {
		id, err := st.CreateJenisBel(ctx, j.nama, j.warna)
		if err != nil {
			return fmt.Errorf("buat jenis bel %s: %w", j.nama, err)
		}
		var audioIDs []int64
		for _, nama := range j.audio {
			audioIDs = append(audioIDs, ids[nama])
		}
		if err := st.SetUrutanBel(ctx, id, audioIDs); err != nil {
			return fmt.Errorf("atur suara %s: %w", j.nama, err)
		}
	}
	// Profil dibuat terakhir: keberadaan profil menandai data awal sudah lengkap.
	profil, err := st.CreateProfil(ctx, "Normal")
	if err != nil {
		return err
	}
	return st.SetSetting(ctx, store.KunciProfilDefault, strconv.FormatInt(profil, 10))
}

// Command belsekolah menjalankan bel sekolah otomatis.
package main

import (
	"context"
	"flag"
	"fmt"
	"log/slog"
	"os"
	"os/signal"
	"path/filepath"
	"syscall"
	"time"

	"belsekolah/internal/bawaan"
	"belsekolah/internal/nada"
	"belsekolah/internal/player"
	"belsekolah/internal/scheduler"
	"belsekolah/internal/store"
)

var _ scheduler.Sumber = (*store.Store)(nil)

func main() {
	dataDir := flag.String("data", "/var/lib/belsekolah", "folder data (database dan audio)")
	flag.Parse()

	log := slog.New(slog.NewTextHandler(os.Stderr, nil))
	if err := jalankan(*dataDir, log); err != nil {
		log.Error("aplikasi berhenti", "err", err)
		os.Exit(1)
	}
}

func jalankan(dataDir string, log *slog.Logger) error {
	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()

	audioDir := filepath.Join(dataDir, "audio")
	if err := os.MkdirAll(audioDir, 0o755); err != nil {
		return err
	}
	// Nada cadangan ditulis ulang setiap start, terpisah dari folder audio.
	cadangan := filepath.Join(dataDir, "cadangan.wav")
	if err := os.WriteFile(cadangan, nada.Cadangan(), 0o644); err != nil {
		return fmt.Errorf("tulis nada cadangan: %w", err)
	}

	st, err := store.Open(filepath.Join(dataDir, "bel.db"))
	if err != nil {
		return err
	}
	defer st.Close()

	if err := bawaan.Pasang(ctx, st, audioDir, time.Now()); err != nil {
		return fmt.Errorf("pasang data awal: %w", err)
	}

	pemutar := player.New(player.Mpv{}, cadangan)
	go pemutar.Jalankan(ctx)

	sch := scheduler.New(scheduler.JamSistem{}, st, pemutar, audioDir, log)
	log.Info("bel sekolah berjalan", "data", dataDir, "zona_waktu", time.Now().Format("MST -07:00"))
	sch.Run(ctx)
	log.Info("bel sekolah berhenti")
	return nil
}
