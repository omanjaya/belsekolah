# Bel Sekolah — Rencana 3: TTS (Teks ke Suara) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Admin bisa mengetik teks, mendengar pratinjau, menyimpannya sebagai suara MP3 berbahasa Indonesia (Gadis/Ardi), dan membuat ulang suara itu setelah teks diedit.

**Architecture:** Paket `tts` memanggil CLI `edge-tts` (Python, via `pipx`) dengan teks lewat file sementara, lalu membaca MP3 hasilnya. API menambah 4 endpoint dan menyimpan hasil sebagai audio `sumber=tts` dengan nama file `tts-<hex>.mp3`. Saat bel berbunyi tidak ada akses internet.

**Tech Stack:** Go 1.27, `edge-tts` CLI (Python, pipx), Docker dev image Debian trixie.

**Spec:** `docs/superpowers/specs/2026-10-15-bel-sekolah-design.md` §8

## Global Constraints

- Module `belsekolah`; tanpa cgo; teks pengguna berbahasa Indonesia.
- Suara yang dikenal: `id-ID-GadisNeural` ("Gadis (perempuan)") dan `id-ID-ArdiNeural` ("Ardi (laki-laki)").
- Teks TTS di-trim, 1–1000 karakter.
- Gagal karena jaringan/layanan → HTTP 503 `"gagal membuat suara TTS; pastikan mini PC terhubung ke internet"`; CLI tidak ada → 503 `"program edge-tts belum terpasang"`.
- Batas waktu satu pembuatan suara: 60 detik.
- Path file audio selalu `filepath.Join(AudioDir, filepath.Base(a.File))`; file ditulis dulu ke file sementara berawalan titik di `AudioDir`, lalu di-rename.
- Setiap perubahan data sukses memanggil `s.ubahData()`.

```go
// Package tts mengubah teks menjadi MP3 dengan memanggil program edge-tts.
// CLI resmi dipakai (bukan protokol sendiri) karena Microsoft sering mengubah
// token DRM-nya; cukup `pipx upgrade edge-tts` saat itu terjadi.
package tts

import (
	"context"
	"errors"
	"fmt"
	"io/fs"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
	"time"
)

type Suara struct {
	ID   string `json:"id"`
	Nama string `json:"nama"`
}

var DaftarSuara = []Suara{
	{ID: "id-ID-GadisNeural", Nama: "Gadis (perempuan)"},
	{ID: "id-ID-ArdiNeural", Nama: "Ardi (laki-laki)"},
}

var (
	ErrSuaraTidakDikenal = errors.New("suara tidak dikenal")
	ErrTidakTerpasang    = errors.New("program edge-tts belum terpasang")
	ErrGagal             = errors.New("gagal membuat suara TTS; pastikan mini PC terhubung ke internet")
)

func SuaraDikenal(id string) bool {
	for _, s := range DaftarSuara {
		if s.ID == id {
			return true
		}
	}
	return false
}

// EdgeTTS menjalankan CLI edge-tts. Nilai kosong memakai "edge-tts" dan batas 60 detik.
type EdgeTTS struct {
	Perintah string
	Batas    time.Duration
}

func (e EdgeTTS) Buat(ctx context.Context, teks, suara string) ([]byte, error) {
	if !SuaraDikenal(suara) {
		return nil, ErrSuaraTidakDikenal
	}
	perintah := e.Perintah
	if perintah == "" {
		perintah = "edge-tts"
	}
	batas := e.Batas
	if batas == 0 {
		batas = 60 * time.Second
	}
	ctx, cancel := context.WithTimeout(ctx, batas)
	defer cancel()

	dir, err := os.MkdirTemp("", "belsekolah-tts-*")
	if err != nil {
		return nil, err
	}
	defer os.RemoveAll(dir)
	fileTeks := filepath.Join(dir, "teks.txt")
	if err := os.WriteFile(fileTeks, []byte(teks), 0o600); err != nil {
		return nil, err
	}
	fileMP3 := filepath.Join(dir, "suara.mp3")

	cmd := exec.CommandContext(ctx, perintah, "--voice", suara, "--file", fileTeks, "--write-media", fileMP3)
	cmd.WaitDelay = 2 * time.Second
	out, err := cmd.CombinedOutput()
	if err != nil {
		if errors.Is(err, exec.ErrNotFound) || errors.Is(err, fs.ErrNotExist) {
			return nil, ErrTidakTerpasang
		}
		if ctx.Err() != nil {
			return nil, fmt.Errorf("%w: melewati batas waktu %s", ErrGagal, batas)
		}
		return nil, fmt.Errorf("%w: %s", ErrGagal, ringkas(out))
	}
	b, err := os.ReadFile(fileMP3)
	if err != nil || len(b) == 0 {
		return nil, fmt.Errorf("%w: edge-tts tidak menghasilkan suara", ErrGagal)
	}
	return b, nil
}

// ringkas mengambil baris terakhir output yang tidak kosong (biasanya pesan error Python).
func ringkas(out []byte) string {
	baris := strings.Split(strings.TrimSpace(string(out)), "\n")
	s := strings.TrimSpace(baris[len(baris)-1])
	if len(s) > 200 {
		s = s[len(s)-200:]
	}
	return s
}
```

```go
const batasTeksTTS = 1000

type permintaanTTS struct {
	Nama  string `json:"nama"`
	Teks  string `json:"teks"`
	Suara string `json:"suara"`
}

func (s *server) listSuaraTTS(w http.ResponseWriter, r *http.Request) {
	tulisJSON(w, http.StatusOK, tts.DaftarSuara)
}

// bacaPermintaanTTS membaca dan memvalidasi body; menulis 400 dan mengembalikan false jika tidak valid.
func bacaPermintaanTTS(w http.ResponseWriter, r *http.Request) (permintaanTTS, bool) {
	var p permintaanTTS
	if !bacaJSON(w, r, &p) {
		return p, false
	}
	p.Teks = strings.TrimSpace(p.Teks)
	switch {
	case p.Teks == "":
		tulisError(w, http.StatusBadRequest, "teks tidak boleh kosong")
		return p, false
	case utf8.RuneCountInString(p.Teks) > batasTeksTTS:
		tulisError(w, http.StatusBadRequest, "teks maksimal 1000 karakter")
		return p, false
	case !tts.SuaraDikenal(p.Suara):
		tulisError(w, http.StatusBadRequest, "suara tidak dikenal")
		return p, false
	}
	return p, true
}

// buatSuara memanggil TTS; menulis respons error dan mengembalikan false jika gagal.
func (s *server) buatSuara(w http.ResponseWriter, r *http.Request, p permintaanTTS) ([]byte, bool) {
	if s.TTS == nil {
		tulisError(w, http.StatusServiceUnavailable, tts.ErrTidakTerpasang.Error())
		return nil, false
	}
	b, err := s.TTS.Buat(r.Context(), p.Teks, p.Suara)
	switch {
	case err == nil:
		return b, true
	case errors.Is(err, tts.ErrTidakTerpasang):
		tulisError(w, http.StatusServiceUnavailable, tts.ErrTidakTerpasang.Error())
	case errors.Is(err, tts.ErrGagal):
		s.Log.Warn("TTS gagal", "err", err)
		tulisError(w, http.StatusServiceUnavailable, tts.ErrGagal.Error())
	case errors.Is(err, tts.ErrSuaraTidakDikenal):
		tulisError(w, http.StatusBadRequest, "suara tidak dikenal")
	default:
		s.tulisErrorData(w, err)
	}
	return nil, false
}

func (s *server) pratinjauTTS(w http.ResponseWriter, r *http.Request) {
	p, ok := bacaPermintaanTTS(w, r)
	if !ok {
		return
	}
	b, ok := s.buatSuara(w, r, p)
	if !ok {
		return
	}
	w.Header().Set("Content-Type", "audio/mpeg")
	w.Header().Set("Cache-Control", "no-store")
	w.Write(b)
}

// simpanSementara menulis data ke file sementara di folder audio dan mengukur durasinya.
// Pemanggil wajib me-rename atau menghapus file yang dikembalikan.
func (s *server) simpanSementara(r *http.Request, data []byte) (string, int64, error) {
	tmp, err := os.CreateTemp(s.AudioDir, ".tts-*")
	if err != nil {
		return "", 0, err
	}
	_, err = tmp.Write(data)
	if cerr := tmp.Close(); err == nil {
		err = cerr
	}
	if err != nil {
		os.Remove(tmp.Name())
		return "", 0, err
	}
	durasi, err := s.DurasiAudio(r.Context(), tmp.Name())
	if err != nil {
		os.Remove(tmp.Name())
		return "", 0, err
	}
	return tmp.Name(), durasi, nil
}

func namaFileTTS() string {
	b := make([]byte, 8)
	if _, err := rand.Read(b); err != nil {
		panic(err)
	}
	return "tts-" + hex.EncodeToString(b) + ".mp3"
}

func (s *server) buatAudioTTS(w http.ResponseWriter, r *http.Request) {
	p, ok := bacaPermintaanTTS(w, r)
	if !ok {
		return
	}
	nama := strings.TrimSpace(p.Nama)
	if nama == "" {
		nama = p.Teks
		if rs := []rune(nama); len(rs) > 40 {
			nama = string(rs[:40])
		}
	}
	data, ok := s.buatSuara(w, r, p)
	if !ok {
		return
	}
	tmp, durasi, err := s.simpanSementara(r, data)
	if err != nil {
		s.tulisErrorData(w, err)
		return
	}
	namaFile := namaFileTTS()
	tujuan := filepath.Join(s.AudioDir, namaFile)
	if err := os.Rename(tmp, tujuan); err != nil {
		os.Remove(tmp)
		s.tulisErrorData(w, err)
		return
	}
	id, err := s.Store.CreateAudio(r.Context(), store.Audio{
		Nama: nama, Sumber: "tts", File: namaFile, DurasiMS: durasi,
		TeksTTS: p.Teks, SuaraTTS: p.Suara, Dibuat: s.Now(),
	})
	if err != nil {
		os.Remove(tujuan)
		s.tulisErrorData(w, err)
		return
	}
	s.ubahData()
	s.kirimAudio(w, r, id, http.StatusCreated)
}

func (s *server) ubahAudioTTS(w http.ResponseWriter, r *http.Request) {
	id, ok := idDariPath(w, r)
	if !ok {
		return
	}
	a, err := s.Store.GetAudio(r.Context(), id)
	if err != nil {
		s.tulisErrorData(w, err)
		return
	}
	if a.Sumber != "tts" {
		tulisError(w, http.StatusConflict, "hanya suara TTS yang bisa dibuat ulang dari teks")
		return
	}
	p, ok := bacaPermintaanTTS(w, r)
	if !ok {
		return
	}
	data, ok := s.buatSuara(w, r, p)
	if !ok {
		return
	}
	tmp, durasi, err := s.simpanSementara(r, data)
	if err != nil {
		s.tulisErrorData(w, err)
		return
	}
	if err := os.Rename(tmp, s.pathAudio(a)); err != nil {
		os.Remove(tmp)
		s.tulisErrorData(w, err)
		return
	}
	if err := s.Store.UpdateAudioTTS(r.Context(), id, p.Teks, p.Suara, durasi); err != nil {
		s.tulisErrorData(w, err)
		return
	}
	s.ubahData()
	s.kirimAudio(w, r, id, http.StatusOK)
}
```
