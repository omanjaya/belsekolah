# Bel Sekolah — Rencana 4: Frontend Svelte Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Antarmuka web berbahasa Indonesia, mobile-first, untuk semua fitur di spec §9: login + ganti password awal, Beranda, Jadwal, Profil & Periode, Libur, Jenis Bel, Pustaka Suara (unggah + TTS), Riwayat, Pengaturan (volume, password, sistem, backup/restore). Hasil build tertanam di binary Go.

**Architecture:** Svelte 5 (runes) + TypeScript + Vite + Tailwind v4 di folder `web/`. Router sederhana berbasis History API (server sudah mengembalikan `index.html` untuk path non-file). Klien API bertipe di `src/lib/api.ts`. Hasil `vite build` ditulis ke `internal/webui/dist` dan **di-commit** (agar `go build` tidak butuh Node). Tes: Vitest untuk logika murni (`src/lib/*.test.ts`), Playwright untuk alur nyata terhadap binary Go sungguhan (dijalankan oleh `webServer` Playwright).

**Tech Stack:** Svelte 5.57, Vite 8, Tailwind CSS 4.3 (`@tailwindcss/vite`), TypeScript, `@lucide/svelte`, Vitest 5, Playwright 1.64 (Chromium), Node 24.

**Spec:** `docs/superpowers/specs/2026-10-15-bel-sekolah-design.md` §9, §14 (kontrak API), §8 (TTS)

## Global Constraints

- Semua teks UI berbahasa Indonesia; pesan error dari API (`{"error": ...}`) ditampilkan apa adanya.
- Mobile-first: target sentuh minimal 44px (`min-h-11`), lebar layar 360px tanpa scroll horizontal; desktop (≥768px) memakai sidebar.
- Jam dan tanggal yang ditampilkan adalah **jam server/sekolah** (diambil dari string ISO server beserta offset-nya), bukan zona HP.
- Setiap aksi yang mengubah data menampilkan notifikasi sukses atau galat; tombol menampilkan status sibuk selama permintaan.
- Hapus selalu lewat dialog konfirmasi.
- Aksesibilitas: setiap input punya `<label>`; dialog memakai `<dialog>`; ikon-saja punya `aria-label`; kontras warna cukup (teks utama `slate-900/800` di atas putih).
- Halaman tidak boleh memakai `localStorage` untuk data; sumber kebenaran adalah API.
- File di `internal/webui/dist` **hanya** di-commit oleh Task 7. Task 1–6 tidak meng-commit hasil build (jalankan `git checkout -- ../internal/webui/dist && git clean -fd ../internal/webui/dist` sebelum commit jika terubah).

`web/src/lib/format.ts`:

```ts
export const NAMA_HARI = ['', 'Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu', 'Minggu']
const BULAN = ['Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni', 'Juli', 'Agustus',
  'September', 'Oktober', 'November', 'Desember']

function urai(tgl: string): [number, number, number] {
  const [y, m, d] = tgl.split('-').map(Number)
  return [y, m, d]
}

export function formatTanggal(tgl: string): string {
  const [y, m, d] = urai(tgl)
  return `${d} ${BULAN[m - 1]} ${y}`
}

/** 1 = Senin … 7 = Minggu, dihitung tanpa zona waktu. */
export function hariDariTanggal(tgl: string): number {
  const [y, m, d] = urai(tgl)
  const w = new Date(Date.UTC(y, m - 1, d)).getUTCDay()
  return w === 0 ? 7 : w
}

export function formatTanggalPanjang(tgl: string): string {
  return `${NAMA_HARI[hariDariTanggal(tgl)]}, ${formatTanggal(tgl)}`
}

export function formatRentang(mulai: string, selesai: string): string {
  return mulai === selesai ? formatTanggal(mulai) : `${formatTanggal(mulai)} – ${formatTanggal(selesai)}`
}

export function formatDurasi(ms: number): string {
  const detik = ms / 1000
  if (detik < 60) {
    return `${detik.toFixed(detik % 1 === 0 ? 0 : 1).replace('.', ',')} dtk`
  }
  const menit = Math.floor(detik / 60)
  return `${menit} mnt ${Math.round(detik - menit * 60)} dtk`
}

export function hitungMundur(ms: number): string {
  if (ms <= 0) return 'sekarang'
  const total = Math.floor(ms / 1000)
  const jam = Math.floor(total / 3600)
  const menit = Math.floor((total % 3600) / 60)
  const detik = total % 60
  if (jam > 0) return `dalam ${jam} jam ${menit} menit`
  if (menit > 0) return `dalam ${menit} menit ${detik} detik`
  return `dalam ${detik} detik`
}

/** Offset zona waktu (menit) dari string ISO server, mis. "+07:00" → 420. */
export function offsetDariISO(iso: string): number {
  const m = iso.match(/([+-])(\d{2}):(\d{2})$/)
  if (!m) return 0
  const nilai = Number(m[2]) * 60 + Number(m[3])
  return m[1] === '-' ? -nilai : nilai
}

function geser(ms: number, offsetMenit: number): Date {
  return new Date(ms + offsetMenit * 60_000)
}

const dua = (n: number) => String(n).padStart(2, '0')

/** HH:MM:SS pada zona server. */
export function formatJam(ms: number, offsetMenit: number): string {
  const t = geser(ms, offsetMenit)
  return `${dua(t.getUTCHours())}:${dua(t.getUTCMinutes())}:${dua(t.getUTCSeconds())}`
}

/** YYYY-MM-DD pada zona server. */
export function tanggalHariIni(ms: number, offsetMenit: number): string {
  const t = geser(ms, offsetMenit)
  return `${t.getUTCFullYear()}-${dua(t.getUTCMonth() + 1)}-${dua(t.getUTCDate())}`
}
```

```ts
export class ApiError extends Error {
  constructor(public status: number, pesan: string) {
    super(pesan)
  }
}

let saatTidakMasuk: () => void = () => {}
/** Dipanggil saat API menjawab 401 (sesi habis). */
export function aturSaatTidakMasuk(f: () => void) {
  saatTidakMasuk = f
}

async function minta<T>(metode: string, path: string, body?: unknown): Promise<T> {
  const headers: Record<string, string> = {}
  const init: RequestInit = { method: metode, headers }
  if (body instanceof FormData) {
    init.body = body
  } else if (body !== undefined) {
    init.body = JSON.stringify(body)
    headers['Content-Type'] = 'application/json'
  }
  let res: Response
  try {
    res = await fetch('/api' + path, init)
  } catch {
    throw new ApiError(0, 'Tidak bisa terhubung ke server bel. Periksa jaringan WiFi.')
  }
  if (res.status === 401 && path !== '/login') saatTidakMasuk()
  if (!res.ok) {
    let pesan = `Terjadi kesalahan (${res.status})`
    try {
      const j = await res.json()
      if (j && typeof j.error === 'string') pesan = j.error
    } catch {
      /* bukan JSON */
    }
    throw new ApiError(res.status, pesan)
  }
  if (res.status === 204) return undefined as T
  const ct = res.headers.get('Content-Type') ?? ''
  if (ct.includes('application/json')) return (await res.json()) as T
  return (await res.blob()) as T
}

export const api = {

}
```

```ts
export const rute = $state({ path: location.pathname })

window.addEventListener('popstate', () => {
  rute.path = location.pathname
})

export function pindah(ke: string) {
  if (ke === location.pathname) return
  history.pushState(null, '', ke)
  rute.path = ke
  window.scrollTo(0, 0)
}

/** Pasang sebagai onclick pada <a href="/..."> agar navigasi tanpa memuat ulang halaman. */
export function tautan(e: MouseEvent) {
  if (e.defaultPrevented || e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return
  const href = (e.currentTarget as HTMLAnchorElement).getAttribute('href')
  if (!href || !href.startsWith('/')) return
  e.preventDefault()
  pindah(href)
}
```

```ts
export interface Notif { id: number; jenis: 'sukses' | 'galat' | 'info'; pesan: string }

export const daftarNotif = $state<Notif[]>([])
let urut = 0

export function beriTahu(pesan: string, jenis: Notif['jenis'] = 'sukses') {
  const id = ++urut
  daftarNotif.push({ id, jenis, pesan })
  setTimeout(() => tutupNotif(id), jenis === 'galat' ? 6000 : 3000)
}

export function tutupNotif(id: number) {
  const i = daftarNotif.findIndex((n) => n.id === id)
  if (i >= 0) daftarNotif.splice(i, 1)
}

export function galat(e: unknown) {
  beriTahu(e instanceof Error ? e.message : String(e), 'galat')
}
```

```ts
interface Permintaan {
  buka: boolean
  judul: string
  pesan: string
  label: string
  jawab: ((ya: boolean) => void) | null
}

export const permintaanKonfirmasi = $state<Permintaan>({ buka: false, judul: '', pesan: '', label: 'Hapus', jawab: null })

/** Menampilkan dialog konfirmasi; resolve true jika pengguna menekan tombol aksi. */
export function konfirmasi(judul: string, pesan: string, label = 'Hapus'): Promise<boolean> {
  permintaanKonfirmasi.jawab?.(false)
  return new Promise((jawab) => {
    Object.assign(permintaanKonfirmasi, { buka: true, judul, pesan, label, jawab })
  })
}
```

```ts
import { execSync } from 'node:child_process'
import { existsSync, readFileSync } from 'node:fs'
import type { APIRequestContext } from '@playwright/test'

export const PASSWORD_E2E = 'rahasia-e2e-123'

/** Password awal: dari file data lokal, atau dari container Docker jika BEL_URL dipakai. */
export function passwordAwal(): string | null {
  const port = Number(process.env.PORT ?? 18080)
  const file = `.e2e-data-${port}/password-awal.txt`
  if (!process.env.BEL_URL) {
    return existsSync(file) ? readFileSync(file, 'utf8').trim() : null
  }
  try {
    return execSync('docker compose exec -T belsekolah cat /var/lib/belsekolah/password-awal.txt', { cwd: '..' })
      .toString().trim()
  } catch {
    return null
  }
}

export async function apiMasuk(request: APIRequestContext) {
  const res = await request.post('/api/login', { data: { password: PASSWORD_E2E } })
  if (!res.ok()) throw new Error(`login API gagal: ${res.status()}`)
}

export async function jenisBelId(request: APIRequestContext, nama: string): Promise<number> {
  const res = await request.get('/api/jenis-bel')
  const daftar = (await res.json()) as { id: number; nama: string }[]
  const j = daftar.find((x) => x.nama === nama)
  if (!j) throw new Error(`jenis bel ${nama} tidak ada`)
  return j.id
}

export async function profilDefault(request: APIRequestContext): Promise<number> {
  const res = await request.get('/api/pengaturan')
  return ((await res.json()) as { profilDefaultId: number }).profilDefaultId
}

/** Nama unik per jalankan tes, agar data antar-tes tidak bentrok. */
export function unik(awalan: string): string {
  return `${awalan} ${Date.now().toString(36).slice(-5)}`
}
```
